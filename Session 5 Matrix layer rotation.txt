class Result {

    public static void matrixRotation(List<List<Integer>> matrix, int r) {
        int m = matrix.size();
        int n = matrix.get(0).size();
        
        int[][] result = new int[m][n];
        int layers = Math.min(m, n) / 2;

        for (int layer = 0; layer < layers; layer++) {
            List<Integer> elements = new ArrayList<>();
            List<int[]> coordinates = new ArrayList<>();

            int top = layer;
            int bottom = m - 1 - layer;
            int left = layer;
            int right = n - 1 - layer;
            for (int i = top; i <= bottom; i++) {
                elements.add(matrix.get(i).get(left));
                coordinates.add(new int[]{i, left});
            }
            for (int j = left + 1; j <= right; j++) {
                elements.add(matrix.get(bottom).get(j));
                coordinates.add(new int[]{bottom, j});
            }
            for (int i = bottom - 1; i >= top; i--) {
                elements.add(matrix.get(i).get(right));
                coordinates.add(new int[]{i, right});
            }
            for (int j = right - 1; j > left; j--) {
                elements.add(matrix.get(top).get(j));
                coordinates.add(new int[]{top, j});
            }

            int total = elements.size();
            int rot = r % total;

            for (int i = 0; i < total; i++) {
                int newPos = (i + rot) % total;
                int[] coord = coordinates.get(i);
                result[coord[0]][coord[1]] = elements.get(newPos);
            }
        }

        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                sb.append(result[i][j]).append(j == n - 1 ? "" : " ");
            }
            sb.append("\n");
        }
        System.out.print(sb.toString());
    }

}
