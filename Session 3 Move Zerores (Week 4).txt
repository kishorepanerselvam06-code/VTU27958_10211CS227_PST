class Solution {
    public void moveZeroes(int[] nums) {
        int insertPosition = 0;

for (int i = 0; i < nums.length; i++) {
    if (nums[i] != 0) {
        nums[insertPosition++] = nums[i];
    }
}

while (insertPosition < nums.length) {
    nums[insertPosition++] = 0;
}
        
    }
}

Output:
Example 1:
Input: nums = [0,1,0,3,12]
Output: [1,3,12,0,0]

Example 2:
Input: nums = [0]
Output: [0]
